cmake_minimum_required(VERSION 3.18)
project(ITL612CM5 VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
option(BUILD_GUI "Build the thermal imaging application" ON)
option(BUILD_TOOLS "Build developer benchmarking tools" OFF)
option(BUILD_TESTING "Build developer tests and UI audit entry points" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(USB REQUIRED IMPORTED_TARGET libusb-1.0)
pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
add_library(mars src/mars.cpp)
target_include_directories(mars PUBLIC src)
target_link_libraries(mars PUBLIC PkgConfig::USB Threads::Threads)
add_library(pipeline src/pipeline.cpp src/fusion.cpp)
target_link_libraries(pipeline PUBLIC mars PkgConfig::GLES OpenMP::OpenMP_CXX)
target_compile_options(pipeline PRIVATE -O3 -ffp-contract=off)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
 target_compile_options(pipeline PRIVATE -mcpu=cortex-a76)
endif()
target_compile_definitions(pipeline PRIVATE ITL_SHADER_DIR="${CMAKE_INSTALL_FULL_DATADIR}/itl612-cm5/shaders")
if(BUILD_TESTING OR BUILD_TOOLS)
 target_compile_definitions(pipeline PRIVATE ITL_BUILD_SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
endif()
if(BUILD_TOOLS)
 pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
 add_executable(itl612-bench src/bench.cpp)
 target_link_libraries(itl612-bench PRIVATE pipeline PkgConfig::EGL)
 install(TARGETS itl612-bench RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(BUILD_TESTING)
 enable_testing()
 add_executable(parser-test tests/parser.cpp)
 target_link_libraries(parser-test PRIVATE mars)
 target_compile_options(parser-test PRIVATE -UNDEBUG)
 add_executable(fusion-check tests/fusion_check.cpp tests/fusion_baseline.cpp)
 target_link_libraries(fusion-check PRIVATE pipeline)
 target_compile_options(fusion-check PRIVATE -O3 -ffp-contract=off)
 if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
  target_compile_options(fusion-check PRIVATE -mcpu=cortex-a76)
 endif()
endif()
if(BUILD_GUI)
 find_package(Qt6 REQUIRED COMPONENTS Widgets Gui OpenGLWidgets)
 add_executable(itl612-cm5 src/app.cpp)
 target_link_libraries(itl612-cm5 PRIVATE pipeline Qt6::Widgets Qt6::Gui Qt6::OpenGLWidgets)
 if(BUILD_TESTING)
  target_compile_definitions(itl612-cm5 PRIVATE ITL_ENABLE_DIAGNOSTICS=1)
  add_executable(measurement-check tests/measurement_check.cpp)
  target_include_directories(measurement-check PRIVATE src)
  target_link_libraries(measurement-check PRIVATE Qt6::Widgets Qt6::Gui)
  target_compile_options(measurement-check PRIVATE -UNDEBUG)
  add_executable(language-check tests/language_check.cpp)
  target_include_directories(language-check PRIVATE src)
  target_link_libraries(language-check PRIVATE Qt6::Widgets Qt6::Gui)
  target_compile_options(language-check PRIVATE -UNDEBUG)
  add_test(NAME language COMMAND language-check)
  set_tests_properties(language PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_IM_MODULE=compose")
  add_test(NAME measurements COMMAND measurement-check)
  set_tests_properties(measurements PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_IM_MODULE=compose")
  pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
  add_executable(gl-memory-control EXCLUDE_FROM_ALL tests/gl_memory_control.cpp)
  target_link_libraries(gl-memory-control PRIVATE Qt6::Widgets Qt6::Gui Qt6::OpenGLWidgets PkgConfig::GLES)
 endif()
 install(TARGETS itl612-cm5 RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
 install(PROGRAMS itl612-start DESTINATION ${CMAKE_INSTALL_BINDIR})
 install(FILES itl612-cm5.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
 install(FILES assets/itl612-cm5.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/1024x1024/apps)
 install(FILES 70-itl612-mars.rules DESTINATION ${CMAKE_INSTALL_LIBDIR}/udev/rules.d)
 install(FILES README.md README.en.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/itl612-cm5)
endif()
install(DIRECTORY shaders/ DESTINATION ${CMAKE_INSTALL_DATADIR}/itl612-cm5/shaders FILES_MATCHING PATTERN "*.comp")

set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME itl612-cm5)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "ITL612 CM5 contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native ITL612 thermal imaging application for Raspberry Pi CM5")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_SECTION graphics)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "ffmpeg, qt6-qpa-plugins, qt6-wayland, fonts-noto-cjk, xdg-user-dirs")
set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/postinst;${CMAKE_CURRENT_SOURCE_DIR}/packaging/postrm")
set(CPACK_SOURCE_GENERATOR "")
include(CPack)
